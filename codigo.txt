function doGet(e) {
  try {
    var params = e ? e.parameter : {};
    var action = params.action || "get_schedule";
    if (action === "execute_command" || action === "run_code") return handleExecuteCommand(params);
    if (action === "atualizar_escala_5x2") return atualizarAbaEscala5x2();
    if (action === "update_slot") return handleUpdateSlot(params);
    if (action === "update_cell") return handleUpdateCell(params);
    if (action === "update_folga") return handleUpdateFolga(params);
    if (action === "update_ausencia") return handleUpdateAusencia(params);
    if (action === "sincronizar_folgas_feriado") return handleSincronizarFolgasFeriado(params);
    if (action === "batch_update") return handleBatchUpdate(params);
    if (action === "sync_senior_bh") return handleSyncSeniorBh(params);
    if (action === "audit_sheet") return handleAuditSheet(params);
    if (action === "restore_formulas") return handleRestoreSaidaFormulas(params);
    if (action === "get_monthly_history") return handleGetMonthlyHistory(params);
    if (action === "save_daily_to_history") return handleSaveDailyToHistory(params);
    if (action === "rotina_virada_diaria") return handleRotinaViradaDiaria(params);
    return handleGetSchedule(params);
  } catch (err) {
    return jsonResponse({ success: false, error: err.toString() });
  }
}

function doPost(e) {
  try {
    var data = {};
    if (e && e.postData && e.postData.contents) {
      try { data = JSON.parse(e.postData.contents); } catch (errJson) { data = e.parameter || {}; }
    } else if (e && e.parameter) {
      data = e.parameter;
    }
    var action = data.action || "get_schedule";
    if (action === "execute_command" || action === "run_code") return handleExecuteCommand(data);
    if (action === "atualizar_escala_5x2") return atualizarAbaEscala5x2();
    if (action === "update_slot") return handleUpdateSlot(data);
    if (action === "update_cell") return handleUpdateCell(data);
    if (action === "update_folga") return handleUpdateFolga(data);
    if (action === "update_ausencia") return handleUpdateAusencia(data);
    if (action === "sincronizar_folgas_feriado") return handleSincronizarFolgasFeriado(data);
    if (action === "batch_update") return handleBatchUpdate(data);
    if (action === "sync_senior_bh") return handleSyncSeniorBh(data);
    if (action === "audit_sheet") return handleAuditSheet(data);
    if (action === "restore_formulas") return handleRestoreSaidaFormulas(data);
    if (action === "get_schedule") return handleGetSchedule(data);
    return jsonResponse({ success: false, error: "Ação não reconhecida: " + action });
  } catch (err) {
    return jsonResponse({ success: false, error: err.toString() });
  }
}

function handleExecuteCommand(data) {
  var lock = LockService.getScriptLock();
  lock.tryLock(30000);
  try {
    var code = data.code || data.command;
    if (!code) return jsonResponse({ success: false, error: "Nenhum código fornecido para execução." });
    
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    // Executa dinamicamente o comando enviado remotamente pelo assistente
    var resultado = eval(code);
    SpreadsheetApp.flush();
    return jsonResponse({ success: true, message: "Comando executado com sucesso!", result: resultado });
  } catch (err) {
    return jsonResponse({ success: false, error: err.toString() });
  } finally {
    try { lock.releaseLock(); } catch (e) {}
  }
}

function colLetter(col) {
  var s = "";
  while (col > 0) {
    var m = (col - 1) % 26;
    s = String.fromCharCode(65 + m) + s;
    col = Math.floor((col - 1) / 26);
  }
  return s;
}

function limparHora(v) {
  var s = String(v || "").trim();
  var m = s.match(/^([+-])?(\d{1,3}):(\d{2})(?::\d{2})?$/);
  if (m) {
    var h = m[2].length < 2 ? "0" + m[2] : m[2];
    return (m[1] === "-" ? "-" : "") + h + ":" + m[3];
  }
  return s;
}

function horaParaFracao(txt, permitirNegativo) {
  var s = String(txt || "").trim().replace(",", ":").replace(/h$/i, "");
  var m = s.match(/^([+-])?(\d{1,2})(?::(\d{1,2}))?$/);
  if (!m) return null;
  if (m[1] === "-" && !permitirNegativo) return null;
  var h = parseInt(m[2], 10);
  var min = m[3] ? parseInt(m[3], 10) : 0;
  if (min > 59) return null;
  if (!permitirNegativo && h > 23) return null;
  var total = (h * 60 + min) / 1440;
  return m[1] === "-" ? -total : total;
}

function linhaValida(cfg, linha) {
  for (var b = 0; b < cfg.blocos.length; b++) {
    if (linha >= cfg.blocos[b].ini && linha <= cfg.blocos[b].fim) return true;
  }
  return false;
}

function lerLinha(abaDiaria, cfg, linha) {
  var r = abaDiaria.getRange(linha, cfg.colDepto, 1, 8).getDisplayValues()[0];
  return {
    departamento: String(r[0] || "").trim(),
    nome: String(r[1] || "").trim(),
    escala: String(r[2] || "").trim(),
    entrada: limparHora(r[3]),
    inicioAlmoco: limparHora(r[4]),
    retornoAlmoco: limparHora(r[5]),
    saida: limparHora(r[6]),
    banco: limparHora(r[7])
  };
}

function handleGetSchedule(params) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var abaDiaria = ss.getSheetByName("Diaria") || ss.getSheetByName("ESCALA DIARIA");
  var abaBanco = ss.getSheetByName("Banco de Dados");
  var abaSenior = ss.getSheetByName("Senior_BH");
  if (!abaDiaria) return jsonResponse({ success: false, error: "Aba 'Diaria' não encontrada na planilha." });

  var resultadoDias = {};
  for (var dia in TF_DIAS) {
    var cfg = TF_DIAS[dia];
    var isDomingo = (dia === "Domingo");
    var valData = abaDiaria.getRange(cfg.celulaData).getValue();
    var dataFormatada = "";
    if (valData) {
      dataFormatada = (valData instanceof Date)
        ? Utilities.formatDate(valData, Session.getScriptTimeZone(), "dd/MM/yyyy")
        : String(valData);
    }

    var slots = [];
    for (var b = 0; b < cfg.blocos.length; b++) {
      var bloco = cfg.blocos[b];
      var qtd = bloco.fim - bloco.ini + 1;
      var range = abaDiaria.getRange(bloco.ini, cfg.colDepto, qtd, 8);
      var vals = range.getDisplayValues();
      var cores = abaDiaria.getRange(bloco.ini, cfg.colDepto + 1, qtd, 1).getBackgrounds();

      for (var i = 0; i < qtd; i++) {
        var r = vals[i];
        var depto = String(r[0] || "").trim();
        var nome = String(r[1] || "").trim();
        var escala = String(r[2] || "").trim();
        if (nome.toUpperCase() === "NOME" || depto.toUpperCase() === "DEPARTAMENTO") continue;
        if (depto === "" && nome === "") continue;

        var corFundo = String(cores[i][0] || "").toLowerCase();
        var isVago = (nome === "" || nome.toUpperCase() === "X" || corFundo === "#ea4335" || corFundo === "#ff0000");

        slots.push({
          linha: bloco.ini + i,
          colunaLetra: colLetter(cfg.colDepto + 1),
          grupo: bloco.grupo,
          departamento: depto,
          nome: (nome.toUpperCase() === "X") ? "" : nome,
          escala: escala,
          horario: isDomingo ? depto : escala,
          entrada: limparHora(r[3]),
          inicioAlmoco: limparHora(r[4]),
          retornoAlmoco: limparHora(r[5]),
          saida: limparHora(r[6]),
          banco: limparHora(r[7]),
          isVago: isVago,
          corFundo: corFundo
        });
      }
    }

    // Leitura completa de Folgas (linhas 6 a 12)
    var rangeFolgas = abaDiaria.getRange(6, cfg.colDepto, 7, 3).getDisplayValues();
    var folgas = [];
    for (var f = 0; f < rangeFolgas.length; f++) {
      var fNome = String(rangeFolgas[f][0] || "").trim();
      var fDept = String(rangeFolgas[f][1] || "").trim();
      var fSol = String(rangeFolgas[f][2] || "").trim();
      if (fNome !== "" || fDept !== "") {
        folgas.push({
          linha: 6 + f,
          departamento: fDept,
          nome: fNome,
          solucao: fSol,
          tipo: "folga"
        });
      }
    }

    // Leitura completa de Ausências / Atestados (linhas 6 a 12)
    var rangeAus = abaDiaria.getRange(6, cfg.colDepto + 3, 7, 4).getDisplayValues();
    var ausenciasCompletas = [];
    var ausenciasNomes = [];
    for (var a = 0; a < rangeAus.length; a++) {
      var aNome = String(rangeAus[a][0] || "").trim();
      var aDept = String(rangeAus[a][1] || "").trim();
      var aSol = String(rangeAus[a][3] || "").trim();
      if (aNome !== "" || aDept !== "") {
        ausenciasCompletas.push({
          linha: 6 + a,
          nome: aNome,
          departamento: aDept,
          solucao: aSol,
          tipo: "ausencia"
        });
        if (aNome !== "") ausenciasNomes.push(aNome);
      }
    }

    // Lista unificada de todos ausentes / de folga no dia para filtros
    for (var k = 0; k < folgas.length; k++) {
      if (folgas[k].nome && ausenciasNomes.indexOf(folgas[k].nome) === -1) {
        ausenciasNomes.push(folgas[k].nome);
      }
    }

    resultadoDias[dia] = {
      diaSemana: dia,
      data: dataFormatada,
      celulaData: cfg.celulaData,
      colunaColab: colLetter(cfg.colDepto + 1),
      colunaAusencia: cfg.colunaAusencia,
      temHorarios: true,
      slots: slots,
      folgas: folgas,
      ausenciasCompletas: ausenciasCompletas,
      ausencias: ausenciasNomes
    };
  }

  var listaColaboradores = [];
  if (abaBanco) {
    var ult = abaBanco.getLastRow();
    if (ult >= 2) {
      var dadosBanco = abaBanco.getRange(2, 1, ult - 1, 5).getValues();
      for (var k = 0; k < dadosBanco.length; k++) {
        var bNome = String(dadosBanco[k][0] || "").trim();
        if (bNome !== "") {
          listaColaboradores.push({
            nome: bNome,
            escala: String(dadosBanco[k][1] || "").trim(),
            folga: String(dadosBanco[k][2] || "").trim(),
            cracha: String(dadosBanco[k][3] || "").trim(),
            departamento: String(dadosBanco[k][4] || "").trim()
          });
        }
      }
    }
  }

  var saldosSenior = [];
  if (abaSenior) {
    var ultS = abaSenior.getLastRow();
    if (ultS >= 2) {
      var dadosSenior = abaSenior.getRange(2, 1, ultS - 1, 4).getDisplayValues();
      for (var s = 0; s < dadosSenior.length; s++) {
        var sCracha = String(dadosSenior[s][0] || "").trim();
        var sNome = String(dadosSenior[s][1] || "").trim();
        var sSaldo = String(dadosSenior[s][2] || "").trim();
        if (sCracha !== "" || sNome !== "") {
          saldosSenior.push({ cracha: sCracha, nome: sNome, saldo: sSaldo });
        }
      }
    }
  }

  return jsonResponse({
    success: true,
    versao: TF_CONFIG.versao,
    dias: resultadoDias,
    colaboradores: listaColaboradores,
    seniorBh: saldosSenior,
    atualizadoEm: Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss")
  });
}

function handleUpdateSlot(params) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var abaDiaria = ss.getSheetByName("Diaria") || ss.getSheetByName("ESCALA DIARIA");
    if (!abaDiaria) return jsonResponse({ success: false, error: "Aba 'Diaria' não encontrada." });

    var dia = params.dia || params.day;
    var cfg = TF_DIAS[dia];
    var linha = parseInt(params.linha || params.row, 10);
    var novoNome = String(params.nome !== undefined ? params.nome : (params.name !== undefined ? params.name : "")).trim();
    if (!cfg || isNaN(linha) || !linhaValida(cfg, linha)) {
      return jsonResponse({ success: false, error: "Parâmetros inválidos: dia ou linha fora do intervalo." });
    }

    var celulaAlvo = (params.colunaLetra && String(params.colunaLetra).trim() !== "")
      ? abaDiaria.getRange(String(params.colunaLetra).trim().toUpperCase() + linha)
      : abaDiaria.getRange(linha, cfg.colDepto + 1);

    if (novoNome === "" || novoNome.toUpperCase() === "X") {
      celulaAlvo.setValue(novoNome === "" ? "" : "X");
      celulaAlvo.setBackground("#ea4335");
    } else {
      celulaAlvo.setValue(novoNome);
      celulaAlvo.setBackground(null);
    }
    celulaAlvo.setFontColor(null);
    SpreadsheetApp.flush();

    return jsonResponse({
      success: true,
      message: "Escala atualizada com sucesso!",
      dia: dia,
      linha: linha,
      colunaLetra: colLetter(cfg.colDepto + 1),
      nome: novoNome,
      linhaAtualizada: lerLinha(abaDiaria, cfg, linha),
      atualizadoEm: Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss")
    });
  } finally {
    try { lock.releaseLock(); } catch (e) {}
  }
}

function handleUpdateCell(params) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var abaDiaria = ss.getSheetByName("Diaria") || ss.getSheetByName("ESCALA DIARIA");
    if (!abaDiaria) return jsonResponse({ success: false, error: "Aba 'Diaria' não encontrada." });

    var dia = params.dia;
    var cfg = TF_DIAS[dia];
    var linha = parseInt(params.linha, 10);
    var campo = String(params.campo || "");
    var valor = String(params.valor !== undefined ? params.valor : "").trim();

    if (!cfg || isNaN(linha) || !linhaValida(cfg, linha)) {
      return jsonResponse({ success: false, error: "Dia ou linha inválidos." });
    }
    if (campo === "nome") return handleUpdateSlot({ dia: dia, linha: linha, nome: valor, colunaLetra: params.colunaLetra });
    if (!TF_CAMPOS.hasOwnProperty(campo)) {
      return jsonResponse({ success: false, error: "Campo não editável: " + campo });
    }

    var cel = (params.colunaLetra && String(params.colunaLetra).trim() !== "")
      ? abaDiaria.getRange(String(params.colunaLetra).trim().toUpperCase() + linha)
      : abaDiaria.getRange(linha, cfg.colDepto + TF_CAMPOS[campo]);

    var isBanco = (campo === "banco");
    if (valor === "") {
      cel.clearContent();
    } else {
      var frac = horaParaFracao(valor, isBanco);
      if (frac === null) {
        return jsonResponse({ success: false, error: "Valor inválido: \"" + valor + "\". Use HH:MM" + (isBanco ? " (ex: 2:00 ou -2:00)." : ".") });
      }
      cel.setValue(frac);
      var fmt = String(cel.getNumberFormat() || "");
      if (fmt.toLowerCase().indexOf("h") === -1) {
        cel.setNumberFormat(isBanco ? "[hh]:mm" : "hh:mm");
      }
    }

    SpreadsheetApp.flush();
    return jsonResponse({
      success: true,
      dia: dia,
      linha: linha,
      campo: campo,
      linhaAtualizada: lerLinha(abaDiaria, cfg, linha),
      atualizadoEm: Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss")
    });
  } finally {
    try { lock.releaseLock(); } catch (e) {}
  }
}

function handleBatchUpdate(data) {
  var lock = LockService.getScriptLock();
  lock.tryLock(15000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var abaDiaria = ss.getSheetByName("Diaria") || ss.getSheetByName("ESCALA DIARIA");
    if (!abaDiaria) return jsonResponse({ success: false, error: "Aba 'Diaria' não encontrada." });

    var updates = data.updates || [];
    if (typeof updates === "string") {
      try { updates = JSON.parse(updates); } catch (eU) {}
    }

    var count = 0;
    for (var u = 0; u < updates.length; u++) {
      var item = updates[u];
      var dia = item.dia;
      var cfg = TF_DIAS[dia];
      var linha = parseInt(item.linha, 10);
      if (!cfg || isNaN(linha) || !linhaValida(cfg, linha)) continue;

      if (item.nome !== undefined) {
        var celNome = (item.colunaLetra)
          ? abaDiaria.getRange(String(item.colunaLetra).trim().toUpperCase() + linha)
          : abaDiaria.getRange(linha, cfg.colDepto + 1);
        var novoNome = String(item.nome || "").trim();
        if (novoNome === "" || novoNome.toUpperCase() === "X") {
          celNome.setValue(novoNome === "" ? "" : "X");
          celNome.setBackground("#ea4335");
        } else {
          celNome.setValue(novoNome);
          celNome.setBackground(null);
        }
        celNome.setFontColor(null);
      }

      ["entrada", "inicioAlmoco", "retornoAlmoco", "banco"].forEach(function(cName) {
        if (item[cName] !== undefined) {
          var cel = abaDiaria.getRange(linha, cfg.colDepto + TF_CAMPOS[cName]);
          var vStr = String(item[cName] || "").trim();
          var isBnc = (cName === "banco");
          if (vStr === "") {
            cel.clearContent();
          } else {
            var fr = horaParaFracao(vStr, isBnc);
            if (fr !== null) {
              cel.setValue(fr);
              cel.setNumberFormat(isBnc ? "[hh]:mm" : "hh:mm");
            }
          }
        }
      });
      count++;
    }

    SpreadsheetApp.flush();
    return jsonResponse({
      success: true,
      message: count + " registros atualizados com sucesso em lote!",
      totalUpdated: count,
      atualizadoEm: Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss")
    });
  } finally {
    try { lock.releaseLock(); } catch (e) {}
  }
}

function handleRestoreSaidaFormulas() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var abaDiaria = ss.getSheetByName("Diaria") || ss.getSheetByName("ESCALA DIARIA");
  if (!abaDiaria) return jsonResponse({ success: false, error: "Aba 'Diaria' não encontrada." });

  var rest = 0;
  for (var dia in TF_DIAS) {
    var cfg = TF_DIAS[dia];
    var colEnt = colLetter(cfg.colDepto + TF_CAMPOS.entrada);
    var colIni = colLetter(cfg.colDepto + TF_CAMPOS.inicioAlmoco);
    var colRet = colLetter(cfg.colDepto + TF_CAMPOS.retornoAlmoco);
    var colBnc = colLetter(cfg.colDepto + TF_CAMPOS.banco);
    var colSaiNum = cfg.colDepto + 6;

    for (var b = 0; b < cfg.blocos.length; b++) {
      var bloco = cfg.blocos[b];
      for (var lin = bloco.ini; lin <= bloco.fim; lin++) {
        var celSai = abaDiaria.getRange(lin, colSaiNum);
        var formulaAtual = celSai.getFormula();
        if (!formulaAtual || formulaAtual === "") {
          var f = '=IF(' + colEnt + lin + '="","",(' + colEnt + lin + '+TIME(8,20,0)+(' + colRet + lin + '-' + colIni + lin + ')+' + colBnc + lin + '))';
          celSai.setFormula(f);
          celSai.setNumberFormat("hh:mm");
          rest++;
        }
      }
    }
  }

  SpreadsheetApp.flush();
  return jsonResponse({ success: true, message: "Fórmulas de Saída verificadas!", totalRestauradas: rest });
}

function handleAuditSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var abaDiaria = ss.getSheetByName("Diaria") || ss.getSheetByName("ESCALA DIARIA");
  var alertas = [];

  for (var dia in TF_DIAS) {
    var cfg = TF_DIAS[dia];
    var nomesNoDia = {};

    for (var b = 0; b < cfg.blocos.length; b++) {
      var bloco = cfg.blocos[b];
      var qtd = bloco.fim - bloco.ini + 1;
      var vals = abaDiaria.getRange(bloco.ini, cfg.colDepto, qtd, 8).getDisplayValues();
      var cores = abaDiaria.getRange(bloco.ini, cfg.colDepto + 1, qtd, 1).getBackgrounds();

      for (var i = 0; i < qtd; i++) {
        var lin = bloco.ini + i;
        var depto = String(vals[i][0] || "").trim();
        var nome = String(vals[i][1] || "").trim();
        var corFundo = String(cores[i][0] || "").toLowerCase();

        if (nome === "" || nome.toUpperCase() === "X" || corFundo === "#ea4335") {
          alertas.push({ tipo: "vaga", dia: dia, linha: lin, setor: depto, mensagem: "Posto " + depto + " está vago." });
        } else {
          var nUpper = nome.toUpperCase();
          if (nomesNoDia[nUpper]) {
            alertas.push({ tipo: "duplicidade", dia: dia, linha: lin, nome: nome, mensagem: nome + " está alocado mais de uma vez na " + dia });
          } else {
            nomesNoDia[nUpper] = lin;
          }
        }
      }
    }
  }

  return jsonResponse({ success: true, totalAlertas: alertas.length, alertas: alertas });
}

function menuAuditarHoje() {
  var res = handleAuditSheet();
  var data = JSON.parse(res.getContent());
  var ui = SpreadsheetApp.getUi();
  if (data.totalAlertas === 0) {
    ui.alert("✅ Escala 100% Auditada!", "Nenhum posto vago ou duplicado encontrado.", ui.ButtonSet.OK);
  } else {
    var msg = "Foram encontrados " + data.totalAlertas + " avisos na semana:\n\n";
    data.alertas.slice(0, 8).forEach(function(a) {

    });
    if (data.alertas.length > 8) msg += "\n... e mais " + (data.alertas.length - 8) + " avisos.";
    ui.alert("⚠️ Auditoria da Escala", msg, ui.ButtonSet.OK);
  }
}

function sincronizarFolgasFeriadoSemana(ss) {
  if (!ss) ss = SpreadsheetApp.getActiveSpreadsheet();
  var abaDiaria = ss.getSheetByName("Diaria") || ss.getSheetByName("ESCALA DIARIA");
  var abaBanco = ss.getSheetByName("Banco de Dados");
  if (!abaDiaria) return { success: false, error: "Aba Diaria não encontrada" };

  var deptoMap = {};
  if (abaBanco) {
    var bData = abaBanco.getRange(2, 1, Math.max(abaBanco.getLastRow() - 1, 1), 5).getValues();
    for (var b = 0; b < bData.length; b++) {
      var bNome = String(bData[b][0] || "").trim().toUpperCase();
      if (bNome) deptoMap[bNome] = String(bData[b][4] || "").trim();
    }
  }

  function getDepto(nome) {
    if (!nome) return "Reposição";
    var upper = nome.toUpperCase().trim();
    if (deptoMap[upper]) return deptoMap[upper];
    for (var k in deptoMap) {
      if (k.indexOf(upper) !== -1 || upper.indexOf(k) !== -1) return deptoMap[k];
      var p1 = k.split(" ")[0];
      var p2 = upper.split(" ")[0];
      if (p1.length > 3 && p1 === p2) return deptoMap[k];
    }
    return "Reposição";
  }

  var meses = {
    "janeiro": "01", "fevereiro": "02", "março": "03", "marco": "03",
    "abril": "04", "maio": "05", "junho": "06", "julho": "07",
    "agosto": "08", "setembro": "09", "outubro": "10", "novembro": "11", "dezembro": "12"
  };

  function normalizarData(val) {
    if (!val) return "";
    if (val instanceof Date) return Utilities.formatDate(val, Session.getScriptTimeZone(), "dd/MM/yyyy");
    var s = String(val).trim().toLowerCase();
    var mNum = s.match(/(\d{1,2})\/(\d{1,2})(?:\/(\d{2,4}))?/);
    if (mNum) {
      return mNum[1].padStart(2, "0") + "/" + mNum[2].padStart(2, "0") + "/" + (mNum[3] ? (mNum[3].length === 2 ? "20" + mNum[3] : mNum[3]) : "2026");
    }
    var mExt = s.match(/(\d{1,2})\s+de\s+([a-zçã]+)(?:\s+de\s+(\d{2,4}))?/);
    if (mExt && meses[mExt[2]]) {
      return mExt[1].padStart(2, "0") + "/" + meses[mExt[2]] + "/" + (mExt[3] || "2026");
    }
    return "";
  }

  // 1. Coleta todas as folgas das abas finalizadas (ex: FOLGAS FERIADO 04/10)
  var sheets = ss.getSheets();
  var finalizedFolgas = [];

  for (var i = 0; i < sheets.length; i++) {
    var sName = sheets[i].getName();
    var upper = sName.toUpperCase();
    // Inclui abas de ORGANIZAÇÃO FOLGAS FERIADO e abas finalizadas
    if (upper.indexOf("FOLGA") !== -1 && upper.indexOf("FERIADO") !== -1) {
      var lastRow = sheets[i].getLastRow();
      if (lastRow < 3) continue;
      var d = sheets[i].getRange(1, 1, lastRow, Math.min(sheets[i].getLastColumn(), 5)).getDisplayValues();
      var colNome = 1, colDia = 2, startR = 2;
      for (var r = 0; r < Math.min(d.length, 5); r++) {
        for (var c = 0; c < d[r].length; c++) {
          var cell = d[r][c].toUpperCase();
          if (cell.indexOf("NOME") !== -1) colNome = c;
          if (cell.indexOf("DIA") !== -1 || cell.indexOf("DATA") !== -1) colDia = c;
        }
        if (d[r][colNome] && d[r][colNome].toUpperCase().indexOf("NOME") !== -1) {
          startR = r + 1;
          break;
        }
      }
      for (var k = startR; k < d.length; k++) {
        var n = String(d[k][colNome] || "").trim();
        var dia = normalizarData(d[k][colDia]);
        if (n && n.toUpperCase() !== "NOME" && dia) {
          finalizedFolgas.push({ aba: sName, nome: n, dataNorm: dia });
        }
      }
    }
  }

  // 2. Mapeamento dos dias na Diaria
  var diasConfig = [
    { nome: "Segunda", celData: "D3",  colNome: 5,  colDepto: 6,  colSit: 7,  colSol: 8 },
    { nome: "Terça",   celData: "M3",  colNome: 14, colDepto: 15, colSit: 16, colSol: 17 },
    { nome: "Quarta",  celData: "V3",  colNome: 23, colDepto: 24, colSit: 25, colSol: 26 },
    { nome: "Quinta",  celData: "AE3", colNome: 32, colDepto: 33, colSit: 34, colSol: 35 },
    { nome: "Sexta",   celData: "AN3", colNome: 41, colDepto: 42, colSit: 43, colSol: 44 },
    { nome: "Sábado",  celData: "AW3", colNome: 50, colDepto: 51, colSit: 52, colSol: 53 },
    { nome: "Domingo", celData: "BF3", colNome: 59, colDepto: 60, colSit: 61, colSol: 62 }
  ];

  var relatorio = [];

  for (var d = 0; d < diasConfig.length; d++) {
    var cfg = diasConfig[d];
    var valData = abaDiaria.getRange(cfg.celData).getValue();
    var dataNorm = normalizarData(valData);
    if (!dataNorm) continue;

    var matches = finalizedFolgas.filter(function(f) { return f.dataNorm === dataNorm; });
    if (matches.length === 0) continue;

    var rangeAtuais = abaDiaria.getRange(6, cfg.colNome, 7, 5).getDisplayValues();
    var slots = [];
    for (var s = 0; s < 7; s++) {
      slots.push({
        linha: 6 + s,
        nome: String(rangeAtuais[s][0] || "").trim(),
        depto: String(rangeAtuais[s][1] || "").trim(),
        situacao: String(rangeAtuais[s][2] || "").trim(),
        solucao: String(rangeAtuais[s][3] || "").trim()
      });
    }

    // 1. LIMPEZA: Remove da lista qualquer colaborador com SITUAÇÃO="FERIADO" que pertencia à data anterior e não a esta nova
    for (var s = 0; s < slots.length; s++) {
      var sl = slots[s];
      var isFer = sl.situacao.toUpperCase().indexOf("FERIADO") !== -1 || sl.depto.toUpperCase().indexOf("FERIADO") !== -1;
      if (sl.nome && isFer) {
        var pertence = matches.some(function(m) {
          var n1 = sl.nome.toUpperCase().trim();
          var n2 = m.nome.toUpperCase().trim();
          return n1 === n2 || n1.indexOf(n2) !== -1 || n2.indexOf(n1) !== -1;
        });
        if (!pertence) {
          abaDiaria.getRange(sl.linha, cfg.colNome).setValue("");
          abaDiaria.getRange(sl.linha, cfg.colDepto).setValue("");
          abaDiaria.getRange(sl.linha, cfg.colSit).setValue("");
          if (cfg.colSol) abaDiaria.getRange(sl.linha, cfg.colSol).setValue("");
          relatorio.push(cfg.nome + ": " + sl.nome + " removido (folga pertencia a outra data)");
          sl.nome = "";
          sl.depto = "";
          sl.situacao = "";
          sl.solucao = "";
        }
      }
    }

    // 2. ADIÇÃO: Aloca os colaboradores que pertencem a esta nova data e ainda não estão presentes
    for (var m = 0; m < matches.length; m++) {
      var fer = matches[m];
      var jaAlocado = slots.some(function(sl) {
        if (!sl.nome || sl.nome.trim() === '' || sl.nome === '-') return false;
        var n1 = sl.nome.toUpperCase().trim();
        var n2 = fer.nome.toUpperCase().trim();
        return n1 === n2 || n1.indexOf(n2) !== -1 || n2.indexOf(n1) !== -1;
      });

      if (!jaAlocado) {
        var slotLivre = slots.find(function(sl) { return !sl.nome || sl.nome === "-"; });
        if (slotLivre) {
          slotLivre.nome = fer.nome;
          var pureDepto = getDepto(fer.nome).replace(/\(FERIADO\)/gi, "").replace(/FERIADO/gi, "").trim();
          slotLivre.depto = pureDepto || "Reposição";
          slotLivre.situacao = "FERIADO";
          abaDiaria.getRange(slotLivre.linha, cfg.colNome).setValue(slotLivre.nome);
          abaDiaria.getRange(slotLivre.linha, cfg.colDepto).setValue(slotLivre.depto);
          abaDiaria.getRange(slotLivre.linha, cfg.colSit).setValue("FERIADO");
          relatorio.push(cfg.nome + " (" + dataNorm + "): " + fer.nome + " alocado na Linha " + slotLivre.linha);
        }
      }
    }
  }

  SpreadsheetApp.flush();
  return { success: true, totalFinalized: finalizedFolgas.length, alteracoes: relatorio };
}

/**
 * Salva todos os 7 dias da aba 'Diaria' no 'Historico_Escalas'
 */
function salvarSemanaNoHistorico() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var abaDiaria = ss.getSheetByName("Diaria");
  if (!abaDiaria) return { success: false, error: "Aba Diaria não encontrada" };

  var abaHist = obterOuCriarAbaHistorico(ss);
  var dias = ["Segunda", "Terça", "Quarta", "Quinta", "Sexta", "Sábado", "Domingo"];
  var registros = [];
  var agora = new Date().toISOString();

  // Lê todas as linhas do histórico para evitar duplicatas exatas (Data + Linha + Tipo)
  var lastRow = abaHist.getLastRow();
  var chavesExistentes = {};
  if (lastRow > 1) {
    var dadosChaves = abaHist.getRange(2, 1, lastRow - 1, 1).getValues();
    for (var i = 0; i < dadosChaves.length; i++) {
      if (dadosChaves[i][0]) chavesExistentes[dadosChaves[i][0]] = 2 + i;
    }
  }

  var diasConfig = [
    { nome: "Segunda", celData: "D3", colDepto: 2, colNome: 3, colAusNome: 6, colAusDepto: 7, colAusSit: 8, colAusSol: 9 },
    { nome: "Terça",   celData: "M3", colDepto: 11, colNome: 12, colAusNome: 15, colAusDepto: 16, colAusSit: 17, colAusSol: 18 },
    { nome: "Quarta",  celData: "V3", colDepto: 20, colNome: 21, colAusNome: 24, colAusDepto: 25, colAusSit: 26, colAusSol: 27 },
    { nome: "Quinta",  celData: "AE3", colDepto: 29, colNome: 30, colAusNome: 33, colAusDepto: 34, colAusSit: 35, colAusSol: 36 },
    { nome: "Sexta",   celData: "AN3", colDepto: 38, colNome: 39, colAusNome: 42, colAusDepto: 43, colAusSit: 44, colAusSol: 45 },
    { nome: "Sábado",  celData: "AW3", colDepto: 47, colNome: 48, colAusNome: 51, colAusDepto: 52, colAusSit: 53, colAusSol: 54 },
    { nome: "Domingo", celData: "BF3", colDepto: 56, colNome: 57, colAusNome: 60, colAusDepto: 61, colAusSit: 62, colAusSol: 63 }
  ];

  for (var d = 0; d < diasConfig.length; d++) {
    var cfg = diasConfig[d];
    var valData = abaDiaria.getRange(cfg.celData).getValue();
    var dataNorm = normalizarData(valData);
    if (!dataNorm) continue;

    var partes = dataNorm.split('/');
    var mesAno = partes.length === 3 ? (partes[1] + '/' + partes[2]) : "";

    // 1. Postos da Escala (Linhas 20 a 51)
    var rangeSlots = abaDiaria.getRange(20, cfg.colDepto, 32, 8).getDisplayValues();
    for (var r = 0; r < rangeSlots.length; r++) {
      var numLinha = 20 + r;
      var depto = String(rangeSlots[r][0] || "").trim();
      var nome = String(rangeSlots[r][1] || "").trim();
      var entrada = String(rangeSlots[r][3] || "").trim();
      var saida = String(rangeSlots[r][7] || "").trim();

      if (!depto && !nome) continue;

      var isVago = (!nome || nome === "-" || nome.toUpperCase() === "VAGO" || nome.toUpperCase() === "X");
      var status = isVago ? "VAGO" : "CONFIRMADO";
      var idChave = dataNorm + "_SLOT_" + numLinha;

      registros.push([
        idChave, dataNorm, cfg.nome, mesAno, numLinha,
        depto, depto, nome, entrada, saida, status, "", "ESCALA_POSTO", agora
      ]);
    }

    // 2. Folgas Semanais (Linhas 6 a 12, Cols Folga)
    var rangeFolgas = abaDiaria.getRange(6, cfg.colNome, 7, 3).getDisplayValues();
    for (var f = 0; f < rangeFolgas.length; f++) {
      var numLinhaF = 6 + f;
      var fNome = String(rangeFolgas[f][0] || "").trim();
      var fDepto = String(rangeFolgas[f][1] || "").trim();
      var fSol = String(rangeFolgas[f][2] || "").trim();

      if (fNome && fNome !== "-") {
        var idChaveF = dataNorm + "_FOLGA_" + numLinhaF;
        registros.push([
          idChaveF, dataNorm, cfg.nome, mesAno, numLinhaF,
          fDepto, "Folga Semanal", fNome, "", "", "FOLGA", fSol, "FOLGA_SEMANAL", agora
        ]);
      }
    }

    // 3. Ausências / Atestados / Feriados (Linhas 6 a 12, Cols Ausência)
    var rangeAus = abaDiaria.getRange(6, cfg.colAusNome, 7, 4).getDisplayValues();
    for (var a = 0; a < rangeAus.length; a++) {
      var numLinhaA = 6 + a;
      var aNome = String(rangeAus[a][0] || "").trim();
      var aDepto = String(rangeAus[a][1] || "").trim();
      var aSit = String(rangeAus[a][2] || "").trim();
      var aSol = String(rangeAus[a][3] || "").trim();

      if (aNome && aNome !== "-") {
        var idChaveA = dataNorm + "_AUSENCIA_" + numLinhaA;
        registros.push([
          idChaveA, dataNorm, cfg.nome, mesAno, numLinhaA,
          aDepto, aSit || "Ausência", aNome, "", "", aSit || "AUSENCIA", aSol, "AUSENCIA_ATESTADO", agora
        ]);
      }
    }
  }

  // Grava novos registros ou atualiza existentes em lote
  var novosParaInserir = [];
  for (var k = 0; k < registros.length; k++) {
    var reg = registros[k];
    var chave = reg[0];
    if (chavesExistentes[chave]) {
      // Atualiza linha existente
      var linhaExistente = chavesExistentes[chave];
      abaHist.getRange(linhaExistente, 1, 1, reg.length).setValues([reg]);
    } else {
      novosParaInserir.push(reg);
    }
  }

  if (novosParaInserir.length > 0) {
    var nextRow = abaHist.getLastRow() + 1;
    abaHist.getRange(nextRow, 1, novosParaInserir.length, novosParaInserir[0].length).setValues(novosParaInserir);
  }

  SpreadsheetApp.flush();
  return {
    success: true,
    totalSalvos: registros.length,
    novos: novosParaInserir.length,
    atualizados: registros.length - novosParaInserir.length
  };
}

/**
 * Consulta o Histórico por Mês/Ano (ex: '11/2026')
 */
function handleGetMonthlyHistory(params) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var abaHist = ss.getSheetByName("Historico_Escalas");
  if (!abaHist) {
    return jsonResponse({ success: true, dias: {}, total: 0 });
  }

  var mesAnoFiltro = params.mes_ano || ""; // ex: "11/2026"
  var dataFiltro = params.data || ""; // ex: "26/10/2026"

  var lastRow = abaHist.getLastRow();
  if (lastRow <= 1) {
    return jsonResponse({ success: true, dias: {}, total: 0 });
  }

  var raw = abaHist.getRange(2, 1, lastRow - 1, 14).getDisplayValues();
  var diasMap = {};
  var total = 0;

  for (var i = 0; i < raw.length; i++) {
    var row = raw[i];
    var dData = row[1];
    var dMesAno = row[3];

    if (mesAnoFiltro && dMesAno !== mesAnoFiltro) continue;
    if (dataFiltro && dData !== dataFiltro) continue;

    if (!diasMap[dData]) {
      diasMap[dData] = {
        data: dData,
        diaSemana: row[2],
        mesAno: dMesAno,
        slots: [],
        folgas: [],
        ausencias: []
      };
    }

    var item = {
      id: row[0],
      linha: Number(row[4]),
      departamento: row[5],
      titular: row[6],
      colaborador: row[7],
      entrada: row[8],
      saida: row[9],
      status: row[10],
      solucao: row[11],
      tipoRegistro: row[12]
    };

    if (item.tipoRegistro === "ESCALA_POSTO") {
      diasMap[dData].slots.push(item);
    } else if (item.tipoRegistro === "FOLGA_SEMANAL") {
      diasMap[dData].folgas.push(item);
    } else {
      diasMap[dData].ausencias.push(item);
    }
    total++;
  }

  return jsonResponse({
    success: true,
    total: total,
    filtro: mesAnoFiltro || dataFiltro || "Todos",
    dias: diasMap
  });
}

function handleSaveDailyToHistory(params) {
  var res = salvarSemanaNoHistorico();
  return jsonResponse(res);
}

/**
 * Rotina de Virada Diária (Janela Deslizante / Rolling 7 Days)
 * Pode ser chamada via Gatilho da Meia-Noite ou manualmente
 */
function rotinaViradaDiaria() {
  // 1. Antes de qualquer mudança, salva o estado atual com segurança no histórico
  salvarSemanaNoHistorico();

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var abaDiaria = ss.getSheetByName("Diaria");
  if (!abaDiaria) return { success: false, error: "Aba Diaria não encontrada" };

  var hoje = new Date();
  var diasSemanaNomes = ["Domingo", "Segunda", "Terça", "Quarta", "Quinta", "Sexta", "Sábado"];

  // Configuração das células de data de cada coluna diária
  var celulasData = {
    "Segunda": "D3",
    "Terça":   "M3",
    "Quarta":  "V3",
    "Quinta":  "AE3",
    "Sexta":   "AN3",
    "Sábado":  "AW3",
    "Domingo": "BF3"
  };

  // Calcula as datas dos próximos 7 dias a partir de hoje
  // Ex: Se hoje for Terça 27/10, o bloco de Terça recebe 27/10, Quarta 28/10, ... Segunda próxima recebe 02/11!
  var atualizacoes = [];
  for (var offset = 0; offset < 7; offset++) {
    var d = new Date(hoje.getTime() + (offset * 24 * 60 * 60 * 1000));
    var diaSemanaIndex = d.getDay();
    var nomeDia = diasSemanaNomes[diaSemanaIndex];
    var celula = celulasData[nomeDia];

    if (celula) {
      var diaFmt = (d.getDate() < 10 ? '0' : '') + d.getDate();
      var mesFmt = ((d.getMonth() + 1) < 10 ? '0' : '') + (d.getMonth() + 1);
      var anoFmt = d.getFullYear();
      var novaDataStr = diaFmt + '/' + mesFmt + '/' + anoFmt;

      var valorAtual = abaDiaria.getRange(celula).getValue();
      var dataAtualNorm = normalizarData(valorAtual);

      if (dataAtualNorm !== novaDataStr) {
        abaDiaria.getRange(celula).setValue(novaDataStr);
        atualizacoes.push(nomeDia + " (" + celula + "): " + dataAtualNorm + " -> " + novaDataStr);
      }
    }
  }

  // 2. Após atualizar as datas, sincroniza folgas de feriado com a escala 5x2 se houver feriados no período
  menuSincronizarFolgasFeriado();

  SpreadsheetApp.flush();
  return { success: true, atualizacoes: atualizacoes };
}

/**
 * Cria ou restaura o gatilho automático que roda todo dia entre 00h e 01h
 */
function configurarGatilhoMeiaNoite() {
  var triggers = ScriptApp.getProjectTriggers();
  for (var i = 0; i < triggers.length; i++) {
    if (triggers[i].getHandlerFunction() === "rotinaViradaDiaria") {
      ScriptApp.deleteTrigger(triggers[i]);
    }
  }

  ScriptApp.newTrigger("rotinaViradaDiaria")
    .timeBased()
    .everyDays(1)
    .atHour(0)
    .create();

  SpreadsheetApp.getActiveSpreadsheet().toast("⏰ Gatilho da meia-noite configurado com sucesso!", "TaskFlow");
  return { success: true, message: "Gatilho diário configurado para rodar à meia-noite." };
}

/**
 * Restaura postos, horários, folgas e ausências do Historico_Escalas para a aba Diaria
 */
function restaurarEscalaDoHistoricoNaDiaria(dataAlvo, colDataOuDia) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var abaDiaria = ss.getSheetByName("Diaria") || ss.getSheetByName("ESCALA DIARIA");
  var abaHist = ss.getSheetByName("Historico_Escalas");
  if (!abaDiaria) return { success: false, error: "Aba Diaria não encontrada" };

  // Mapeamento matemático perfeito dos 7 dias com colunas de Folga, Ausência e Slots:
  var diasConfig = {
    4:  { k: 0, nome: "Segunda", celData: "D3",  colData: 4,  colFolga: 2,  colAus: 5,  colSlots: 3 },
    13: { k: 1, nome: "Terça",   celData: "M3",  colData: 13, colFolga: 11, colAus: 14, colSlots: 12 },
    22: { k: 2, nome: "Quarta",  celData: "V3",  colData: 22, colFolga: 20, colAus: 23, colSlots: 21 },
    31: { k: 3, nome: "Quinta",  celData: "AE3", colData: 31, colFolga: 29, colAus: 32, colSlots: 30 },
    40: { k: 4, nome: "Sexta",   celData: "AN3", colData: 40, colFolga: 38, colAus: 41, colSlots: 39 },
    49: { k: 5, nome: "Sábado",  celData: "AW3", colData: 49, colFolga: 47, colAus: 50, colSlots: 48 },
    58: { k: 6, nome: "Domingo", celData: "BF3", colData: 58, colFolga: 56, colAus: 59, colSlots: 57 }
  };

  var cfg = null;
  if (typeof colDataOuDia === "number" && diasConfig[colDataOuDia]) {
    cfg = diasConfig[colDataOuDia];
  } else {
    for (var k in diasConfig) {
      if (diasConfig[k].nome === colDataOuDia) {
        cfg = diasConfig[k];
        break;
      }
    }
  }
  if (!cfg) cfg = diasConfig[4]; // Default: Segunda

  // Normaliza dataAlvo em string dd/MM/yyyy
  var sData = String(dataAlvo || "").trim();
  if (dataAlvo instanceof Date) {
    var dd = dataAlvo.getDate();
    var mm = dataAlvo.getMonth() + 1;
    var yy = dataAlvo.getFullYear();
    sData = (dd < 10 ? "0" + dd : dd) + "/" + (mm < 10 ? "0" + mm : mm) + "/" + yy;
  }

  // 1. VERIFICA SE A DATA CORRESPONDE A OUTRO DIA DA PRÓPRIA SEMANA NA ABA DIARIA (ex: Terça vizinha)
  var diaIrmao = null;
  for (var cK in diasConfig) {
    if (diasConfig[cK].colData === cfg.colData) continue;
    var dataIrmao = abaDiaria.getRange(diasConfig[cK].celData).getDisplayValue().trim();
    if (dataIrmao === sData) {
      diaIrmao = diasConfig[cK];
      break;
    }
  }

  if (diaIrmao) {
    // 1.1 Copia Escala de Reposição (linhas 20 a 51)
    var dOrigem = abaDiaria.getRange(20, diaIrmao.colSlots, 32, 7).getValues();
    var dDestino = abaDiaria.getRange(20, cfg.colSlots, 32, 7).getValues();

    for (var idx = 0; idx < 32; idx++) {
      dDestino[idx][0] = dOrigem[idx][0]; // Nome
      dDestino[idx][1] = dOrigem[idx][1]; // Escala
      dDestino[idx][2] = dOrigem[idx][2]; // Entrada
      dDestino[idx][3] = dOrigem[idx][3]; // IniAlm
      dDestino[idx][4] = dOrigem[idx][4]; // RetAlm
      dDestino[idx][6] = dOrigem[idx][6]; // Banco
    }
    abaDiaria.getRange(20, cfg.colSlots, 32, 7).setValues(dDestino);

    // 1.2 Copia Folgas (linhas 6 a 12, 7 linhas x 3 colunas)
    var fOrigem = abaDiaria.getRange(6, diaIrmao.colFolga, 7, 3).getValues();
    abaDiaria.getRange(6, cfg.colFolga, 7, 3).setValues(fOrigem);

    // 1.3 Copia Atestados/Férias/Feriados (linhas 6 a 12, 7 linhas x 4 colunas)
    var aOrigem = abaDiaria.getRange(6, diaIrmao.colAus, 7, 4).getValues();
    abaDiaria.getRange(6, cfg.colAus, 7, 4).setValues(aOrigem);

    SpreadsheetApp.flush();
    try {
      ss.toast("📥 Escala, Folgas e Atestados de " + sData + " copiados de " + diaIrmao.nome + " para " + cfg.nome + "!", "TaskFlow", 5);
    } catch(e) {}

    return {
      success: true,
      copiadoDe: diaIrmao.nome,
      diaDestino: cfg.nome,
      dataRestaurada: sData,
      slotsRestaurados: 32
    };
  }

  // 2. SE NÃO FOR OUTRO DIA DA SEMANA, BUSCA NO HISTÓRICO MENSAL
  if (!abaHist) return { success: false, error: "Aba Historico_Escalas não encontrada" };
  var lr = abaHist.getLastRow();
  if (lr <= 1) return { success: false, error: "Histórico vazio" };

  var rows = abaHist.getRange(2, 1, lr - 1, 19).getValues();
  var slotsParaRestaurar = {};
  var folgasParaRestaurar = [];
  var ausenciasParaRestaurar = [];

  for (var i = 0; i < rows.length; i++) {
    var r = rows[i];
    var rData = String(r[1] || "").trim();
    if (r[1] instanceof Date) {
      var d = r[1].getDate();
      var m = r[1].getMonth() + 1;
      var y = r[1].getFullYear();
      rData = (d < 10 ? "0" + d : d) + "/" + (m < 10 ? "0" + m : m) + "/" + y;
    }

    if (rData !== sData) continue;

    var tipo = r[16];
    var numLinha = Number(r[4]);

    if (tipo === "ESCALA_POSTO" && numLinha >= 20 && numLinha <= 57) {
      slotsParaRestaurar[numLinha] = {
        nome: String(r[7] || "").trim(),
        entrada: r[8],
        inicioAlmoco: r[9],
        retornoAlmoco: r[10],
        saida: r[11],
        banco: r[12],
        escala: String(r[13] || "").trim()
      };
    } else if (tipo === "FOLGA_SEMANAL") {
      folgasParaRestaurar.push([
        String(r[7] || "").trim(), // Nome
        String(r[5] || "").trim(), // Depto
        String(r[15] || "").trim() // Solução
      ]);
    } else if (tipo === "AUSENCIA_ATESTADO") {
      ausenciasParaRestaurar.push([
        String(r[7] || "").trim(), // Nome
        String(r[5] || "").trim(), // Depto
        String(r[14] || "").trim(), // Situação
        String(r[15] || "").trim()  // Solução
      ]);
    }
  }

  var totalSlots = Object.keys(slotsParaRestaurar).length;
  if (totalSlots === 0 && folgasParaRestaurar.length === 0 && ausenciasParaRestaurar.length === 0) {
    return { success: false, notFound: true, message: "Nenhum registro encontrado no histórico para a data " + sData };
  }

  // 2.1 Preenche Escala de Reposição
  var rangeSlots = abaDiaria.getRange(20, cfg.colSlots, 32, 7);
  var valoresSlots = rangeSlots.getValues();

  for (var rIdx = 0; rIdx < 32; rIdx++) {
    var l = 20 + rIdx;
    if (slotsParaRestaurar[l]) {
      var s = slotsParaRestaurar[l];
      valoresSlots[rIdx][0] = s.nome;
      valoresSlots[rIdx][1] = s.escala;
      valoresSlots[rIdx][2] = s.entrada;
      valoresSlots[rIdx][3] = s.inicioAlmoco;
      valoresSlots[rIdx][4] = s.retornoAlmoco;
      valoresSlots[rIdx][6] = s.banco;
    } else {
      valoresSlots[rIdx][0] = "";
      valoresSlots[rIdx][1] = "";
      valoresSlots[rIdx][2] = "";
      valoresSlots[rIdx][3] = "";
      valoresSlots[rIdx][4] = "";
      valoresSlots[rIdx][6] = "";
    }
  }
  rangeSlots.setValues(valoresSlots);

  // 2.2 Preenche Folgas (7 linhas x 3 colunas)
  var valsFolgas = [
    ["", "", ""], ["", "", ""], ["", "", ""],
    ["", "", ""], ["", "", ""], ["", "", ""], ["", "", ""]
  ];
  for (var f = 0; f < 7 && f < folgasParaRestaurar.length; f++) {
    valsFolgas[f] = folgasParaRestaurar[f];
  }
  abaDiaria.getRange(6, cfg.colFolga, 7, 3).setValues(valsFolgas);

  // 2.3 Preenche Atestados/Férias/Feriados (7 linhas x 4 colunas)
  var valsAus = [
    ["", "", "", ""], ["", "", "", ""], ["", "", "", ""],
    ["", "", "", ""], ["", "", "", ""], ["", "", "", ""], ["", "", "", ""]
  ];
  for (var a = 0; a < 7 && a < ausenciasParaRestaurar.length; a++) {
    valsAus[a] = ausenciasParaRestaurar[a];
  }
  abaDiaria.getRange(6, cfg.colAus, 7, 4).setValues(valsAus);

  SpreadsheetApp.flush();
  try {
    ss.toast("📥 Escala, Folgas e Atestados de " + sData + " restaurados com sucesso em " + cfg.nome + "!", "TaskFlow Histórico", 5);
  } catch(e) {}

  return {
    success: true,
    dataRestaurada: sData,
    diaDestino: cfg.nome,
    slotsRestaurados: totalSlots,
    folgasRestauradas: folgasParaRestaurar.length,
    ausenciasRestauradas: ausenciasParaRestaurar.length
  };
}

function menuPuxarMemoriaHistorico() {
  var ui = SpreadsheetApp.getUi();
  var resp = ui.prompt("📥 Puxar Memória do Histórico", "Digite a data que deseja puxar (ex: 26/10/2026):", ui.ButtonSet.OK_CANCEL);
  if (resp.getSelectedButton() !== ui.Button.OK) return;

  var dataStr = resp.getResponseText().trim();
  if (!dataStr) return;

  var respDia = ui.prompt("Dia da Semana", "Em qual coluna deseja carregar? Digite: Segunda, Terça, Quarta, Quinta, Sexta, Sábado ou Domingo:", ui.ButtonSet.OK_CANCEL);
  if (respDia.getSelectedButton() !== ui.Button.OK) return;

  var diaDestino = respDia.getResponseText().trim();
  var res = restaurarEscalaDoHistoricoNaDiaria(dataStr, diaDestino);

  if (res.success) {
    ui.alert("Memória Restaurada!", "A escala de " + dataStr + " foi restaurada com sucesso na coluna de " + res.diaDestino + "!\n\n• Postos preenchidos: " + res.slotsRestaurados + "\n• Folgas restauradas: " + res.folgasRestauradas, ui.ButtonSet.OK);
  } else {
    ui.alert("Aviso", res.message || res.error || "Data não encontrada no histórico.", ui.ButtonSet.OK);
  }
}
